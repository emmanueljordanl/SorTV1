cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico_w CACHE STRING "Hardware board")
if(NOT DEFINED ENV{PICO_SDK_PATH} AND NOT PICO_SDK_PATH)
  message(FATAL_ERROR "Set PICO_SDK_PATH to pinned SDK 2.3.1 (see README)")
endif()
if(NOT PICO_SDK_PATH)
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(sortv1_firmware C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(sortv1_pico src/main.cpp src/state_machine.cpp src/protocol.cpp src/sensors.cpp src/actuators.cpp vendor/vl53l0x/VL53L0X.cpp)
target_include_directories(sortv1_pico PRIVATE include vendor/shim vendor/vl53l0x)
target_compile_options(sortv1_pico PRIVATE -Wall -Wextra)
target_compile_definitions(sortv1_pico PRIVATE PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0 PICO_STDIO_USB_ENABLE_RESET_VIA_VENDOR_INTERFACE=0)
target_link_libraries(sortv1_pico pico_stdlib pico_rand pico_unique_id hardware_i2c hardware_pwm hardware_timer hardware_watchdog)
pico_enable_stdio_usb(sortv1_pico 1)
pico_enable_stdio_uart(sortv1_pico 0)
pico_set_program_name(sortv1_pico "SorTV1")
pico_set_program_version(sortv1_pico "0.2.0-mvp-rc1")
pico_add_extra_outputs(sortv1_pico)
if(PICO_NO_PICOTOOL)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_custom_command(TARGET sortv1_pico POST_BUILD
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../../scripts/bin_to_uf2.py
      ${CMAKE_CURRENT_BINARY_DIR}/sortv1_pico.bin ${CMAKE_CURRENT_BINARY_DIR}/sortv1_pico.uf2
    BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/sortv1_pico.uf2 VERBATIM)
endif()
